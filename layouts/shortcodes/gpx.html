{{- /* Usage: {{< gpx "route.gpx" >}} or {{< gpx src="route.gpx" caption="Nazaré → Fátima" download=true >}} or {{< gpx src="day-1.gpx, day-2.gpx" labels="Day 1, Day 2" >}} or {{< gpx src="stage-*.gpx" >}} (globs match in name order) (Hugo can't mix positional and named params.) The GPX files must live in the post's page bundle. Several files share one map, each in its own colour with a legend row. Labels default to the GPX's own track name. The "Download GPX" link is off unless download=true. map-only=true renders just the map: no stats, legend or caption. */ -}} {{- $mapOnly := eq (printf "%v" (.Get "map-only")) "true" -}} {{- $src := cond .IsNamedParams (.Get "src") (.Get 0) -}} {{- $files := slice -}} {{- range split $src "," -}} {{- $pattern := trim . " " -}} {{- $matches := $.Page.Resources.Match $pattern -}} {{- if not $matches -}} {{- errorf "gpx shortcode: %q not found in page bundle of %s" $pattern $.Page.File.Path -}} {{- end -}} {{- range sort $matches "Name" }}{{ $files = $files | append . }}{{ end -}} {{- end -}} {{- $labels := slice -}} {{- with .Get "labels" -}} {{- range split . "," }}{{ $labels = $labels | append (trim . " ") }}{{ end -}} {{- if ne (len $labels) (len $files) -}} {{- errorf "gpx shortcode: %d labels for %d files in %s" (len $labels) (len $files) $.Page.File.Path -}} {{- end -}} {{- end -}} {{- $tracks := slice -}} {{- range $i, $f := $files -}} {{- $tracks = $tracks | append (dict "url" $f.RelPermalink "label" (index $labels $i | default "")) -}} {{- end -}} {{- if not (.Page.Store.Get "gpxAssets") -}} {{- .Page.Store.Set "gpxAssets" true -}} {{- $js := resources.Get "js/gpx.js" | js.Build (dict "minify" true) | fingerprint -}} {{- $css := resources.Get "css/gpx.css" | minify | fingerprint -}} {{- end }}
